cmake_minimum_required(VERSION 3.16)
project(camera_host_tests LANGUAGES C CXX)
enable_testing()
add_executable(camera_core_tests test_core.cpp ../main/project_core.cpp)
target_compile_features(camera_core_tests PRIVATE cxx_std_17)
target_compile_options(camera_core_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME camera_core COMMAND camera_core_tests)
add_executable(camera_manager_tests test_camera_manager.cpp ../main/project_camera_manager.cpp)
target_include_directories(camera_manager_tests PRIVATE fakes)
target_compile_features(camera_manager_tests PRIVATE cxx_std_17)
target_compile_options(camera_manager_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME camera_manager COMMAND camera_manager_tests)
add_executable(identity_json_tests test_identity_json.cpp ../main/project_identity_json.cpp ../vendor/cjson/cJSON.c)
target_include_directories(identity_json_tests PRIVATE ../vendor/cjson)
target_compile_features(identity_json_tests PRIVATE cxx_std_17)
add_test(NAME identity_json COMMAND identity_json_tests)
if(MINGW)
  foreach(target camera_core_tests camera_manager_tests identity_json_tests)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
  endforeach()
endif()
